const API = 'https://vrmapi.victronenergy.com/v2/installations/'; const rules = [ ['soc', [/battery.*(state of charge|soc)/i, /^soc$/i]], ['batteryVoltage', [/battery.*voltage/i]], ['batteryCurrent', [/battery.*current/i]], ['batteryPower', [/battery.*power/i, /^dc power$/i]], ['solarPower', [/^(pv charger|solar charger).*power$/i, /^solar power$/i, /^pv power$/i]], ['acInputPower', [/^ac input.*power$/i, /^grid.*power$/i]] ]; function numeric(r) { const x = r.rawValue ?? r.formattedValue; if (typeof x === 'number') { return Number.isFinite(x) ? x : null; } if (typeof x === 'string') { const n = Number(x.trim()); return x.trim() !== '' && Number.isFinite(n) ? n : null; } return null; } function deviceText(r) { return [ r.Device, r.device, r.deviceName, r.deviceType, r.serviceType, r.service, r.instanceName ].filter(x => x != null).join(' '); } function summarize(records) { const values = {}; const selected = {}; for (const r of records) { const name = String(r.description || ''); const value = numeric(r); if (value === null) continue; const device = deviceText(r); const isSmartShunt = /smartshunt/i.test(device); const isBatteryMonitor = /battery monitor|bmv/i.test(device); const priority = isSmartShunt ? 3 : isBatteryMonitor ? 2 : 0; const timestamp = Number(r.timestamp) || 0; for (const [key, patterns] of rules) { if (!patterns.some(p => p.test(name))) continue; const batteryField = [ 'soc', 'batteryVoltage', 'batteryCurrent' ].includes(key); if (batteryField && priority === 0) continue; const old = selected[key]; if ( !old || priority > old.priority || (priority === old.priority && timestamp > old.timestamp) ) { values[key] = value; selected[key] = { priority, timestamp }; } } } return values; } export default async function handler(req, res) { res.setHeader('Cache-Control', 'no-store'); res.setHeader('X-Content-Type-Options', 'nosniff'); if (req.method !== 'GET') { return res.status(405).json({ error: 'Method not allowed' }); } const token = process.env.VRM_ACCESS_TOKEN; const id = process.env.VRM_INSTALLATION_ID; if (!token || !id) { return res.status(503).json({ error: 'VRM configuration incomplete' }); } try { const response = await fetch( `${API}${encodeURIComponent(id)}/diagnostics?count=1000`, { headers: { 'x-authorization': `Token ${token}`, Accept: 'application/json' }, signal: AbortSignal.timeout(12000) } ); if (!response.ok) { return res.status(502).json({ error: 'VRM upstream error', status: response.status }); } const json = await response.json(); const records = Array.isArray(json.records) ? json.records : []; return res.status(200).json({ source: 'Victron VRM', vessel: 'CANTILLN', fetchedAt: new Date().toISOString(), values: summarize(records), note: 'Values are last VRM diagnostics, not guaranteed live.' }); } catch (e) { return res.status(502).json({ error: 'VRM connection failed' }); } }
